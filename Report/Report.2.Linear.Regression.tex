def read_csv(path):
    xs = []
    ys = []
    f = open(path, "r")
    for line in f:
        line = line.strip()
        if line == "":
            continue
        parts = line.split(",")
        xs.append(float(parts[0]))
        ys.append(float(parts[1]))
    f.close()
    return xs, ys

def predict(x, w0, w1):
    return w1 * x + w0

        s = s + e * e
    return 0.5 * s / N

def train(w0, w1, r, iters):
    for it in range(iters):
        for i in range(N):
            e = w1 * xs[i] + w0 - ys[i]
            w0 = w0 - r * e
            w1 = w1 - r * xs[i] * e
        print(f"Iteration {it+1}: w0 = {w0:.4f}, "
              f"w1 = {w1:.4f}, J = {loss(w0, w1):.4f}")
    return w0, w1

The training loop walks through all five data points each iteration and
updates $w_0$ and $w_1$ after every sample.
