\section{Implementation}
% Gradient descent written from scratch in Python. Update rule:
% \[
% x_{n+1} = x_n - r \cdot f'(x_n).
% \]
% For $f(x) = x^2$ we have $f'(x) = 2x$, so the rule becomes
% $x_{n+1} = x_n(1 - 2r)$. The fixed point is $x^*=0$ and convergence
% requires $|1-2r| < 1$, i.e.\ $0 < r < 1$.

\begin{verbatim}
def func(x):
    return x * x

def derivative(x):
    return 2 * x

def gd(start, lr, iter):
    x = start
    for i in range(iter):
        x = x - lr * derivative(x)
        print(f"Iteration {i+1}: x = {x:.4f}, f(x) = {func(x):.4f}")
    return x
\end{verbatim}

\textbf{Comparison: }
With $r=0.1$ in Section 2, $f(x)$ dropped from $64$ to $0$
in only $56$ iterations. With $r=10^{-6}$, after $2{,}300{,}000$
iterations $f(x)$ is still $0.0101$ and $x$ moved from $10$
to $0.1005$. 

\noindent
\textbf{Conclusion.}
A small rate ( $10^{-6}$) does not converge after 2.3 millions iterations. A moderate rate ($0.1$) reaches the
minimum quickly.
